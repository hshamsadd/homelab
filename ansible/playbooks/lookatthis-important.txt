resource "libvirt_network" "vm_network" {
  name      = var.network_name
  autostart = true

  forward = var.network_mode == "nat" ? {
    mode = "nat"
  } : null

  bridge = var.network_mode == "nat" ? {
    name = var.bridge_name
  } : null

      family  = "ipv4"
